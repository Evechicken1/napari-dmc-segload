# napari-dmc-segload Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone napari plugin that scans an animal's registered channels, loads each chosen channel as its own adjustable image layer (plus empty tracing layers for segmentation channels), and saves traced layers in DMC-BrainMap's on-disk format.

**Architecture:** A pip-installable napari plugin (`src/` layout, npe2 manifest) contributing one dock widget. Pure-logic modules (`channel_scanner.py`, `io.py`) are unit-tested against fixture folder trees; the Qt widget (`_widget.py`) wires them to the napari viewer and is covered by a smoke test using `make_napari_viewer`. No dmc files are edited and no dmc runtime internals are used — only dmc's stable on-disk layout.

**Tech Stack:** Python, napari, magicgui/qtpy, numpy, pandas, opencv (cv2), natsort, pytest + pytest-qt.

```python
import json
from pathlib import Path
from typing import List, Optional

from .constants import CANONICAL_CHANNELS


class ChannelScanner:
    """Detect which channels are available for an animal folder."""

    def __init__(self, input_path: Path):
        self.input_path = Path(input_path)

    def _folders_with_tifs(self) -> List[str]:
        sc_dir = self.input_path / "single_channel"
        if not sc_dir.is_dir():
            return []
        found = []
        for sub in sc_dir.iterdir():
            if sub.is_dir() and any(sub.glob("*.tif")):
                found.append(sub.name)
        return found

    def _imaged_channels(self) -> Optional[List[str]]:
        params_fn = self.input_path / "params.json"
        if not params_fn.exists():
            return None
        try:
            data = json.loads(params_fn.read_text())
            return data["general"]["chans_imaged"]
        except (json.JSONDecodeError, KeyError):
            return None

    def available_channels(self) -> List[str]:
        folders = set(self._folders_with_tifs())
        imaged = self._imaged_channels()
        if imaged is not None:
            folders &= set(imaged)
        return [c for c in CANONICAL_CHANNELS if c in folders]
```

from pathlib import Path

import cv2
import numpy as np
from natsort import natsorted


def _channel_tifs(input_path: Path, chan: str):
    chan_dir = Path(input_path) / "single_channel" / chan
    return natsorted([p for p in chan_dir.glob("*.tif")])


def resolve_image(input_path: Path, chan: str, image_idx: int) -> Path:
    """Path to the natsorted image at `image_idx` in single_channel/<chan>/."""
    tifs = _channel_tifs(input_path, chan)
    if image_idx < 0 or image_idx >= len(tifs):
        raise IndexError(
            f"image_idx {image_idx} out of range for channel '{chan}' "
            f"({len(tifs)} images)"
        )
    return tifs[image_idx]


def image_stem(input_path: Path, chan: str, image_idx: int) -> str:
    """Image file name without its .tif extension (dmc's save base name)."""
    return resolve_image(input_path, chan, image_idx).with_suffix("").name


def load_grayscale(path: Path) -> np.ndarray:
    """Load a single-channel image as a 2-D grayscale array."""
    return cv2.imread(str(path), cv2.IMREAD_GRAYSCALE)
```

import pandas as pd


def _seg_dir(input_path: Path, seg_type: str, chan: str) -> Path:
    d = Path(input_path) / "segmentation" / seg_type / chan
    d.mkdir(parents=True, exist_ok=True)
    return d


def _save_name(seg_dir: Path, image_stem: str, seg_type: str) -> Path:
    return seg_dir / f"{image_stem}_{seg_type}.csv"


def save_points_layer(data, input_path: Path, seg_type: str, chan: str, image_stem: str):
    """Write a points layer to dmc's CSV format. Empty data removes a stale file
    and returns None; otherwise returns the written path."""
    seg_dir = _seg_dir(input_path, seg_type, chan)
    save_name = _save_name(seg_dir, image_stem, seg_type)
    arr = np.asarray(data)
    if len(arr) == 0:
        if save_name.exists():
            save_name.unlink()
        return None
    df = pd.DataFrame(arr, columns=["Position Y", "Position X"])
    df.to_csv(save_name)  # default index column, matching dmc
    return save_name


def save_shapes_layer(shapes, input_path: Path, seg_type: str, chan: str, image_stem: str):
    """Write a shapes (polygon) layer to dmc's CSV format with idx_shape."""
    seg_dir = _seg_dir(input_path, seg_type, chan)
    save_name = _save_name(seg_dir, image_stem, seg_type)
    if len(shapes) == 0:
        if save_name.exists():
            save_name.unlink()
        return None
    frames = []
    for i, poly in enumerate(shapes):
        df = pd.DataFrame(np.asarray(poly), columns=["Position Y", "Position X"])
        df["idx_shape"] = i
        frames.append(df)
    pd.concat(frames).to_csv(save_name)
    return save_name
```

        n_probes=dict(widget_type="SpinBox", label="number of fibers/probes",
                      value=1, min=1),
        point_size=dict(widget_type="SpinBox", label="point size", value=5, min=1),
        image_idx=dict(widget_type="SpinBox", label="image index", value=0, min=0),
        call_button=False,
    )
    def form(input_path, reference_channels, segmentation_channels, seg_type,
             n_probes, point_size, image_idx):
        pass
    return form


class SegLoaderWidget(QWidget):
    """Dock widget: scan channels, load per-channel images + tracing layers,
    and save traced layers in dmc's on-disk format."""

    def __init__(self, napari_viewer: Viewer):
        super().__init__()
        self.viewer = napari_viewer
        self.setLayout(QVBoxLayout())

        self.form = _build_form()
        self.layout().addWidget(self.form.native)

        self.btn_scan = QPushButton("Scan channels")
        self.btn_scan.clicked.connect(self._scan_channels)
        self.layout().addWidget(self.btn_scan)

        self.btn_load = QPushButton("Load into viewer")
        self.btn_load.clicked.connect(self._load_into_viewer)
        self.layout().addWidget(self.btn_load)

        self.btn_save = QPushButton("Save segmentation")
        self.btn_save.clicked.connect(self._save_segmentation)
        self.layout().addWidget(self.btn_save)

    # --- helpers -----------------------------------------------------------
    def _input_path(self):
        p = self.form.input_path.value
        if not p or not Path(p).is_dir():
            show_info("Select a valid input path (animal_id) directory.")
            return None
        return Path(p)

    def _selected_channels(self):
        ref = list(self.form.reference_channels.value)
        seg = list(self.form.segmentation_channels.value)
        overlap = set(ref) & set(seg)
        if overlap:
            show_info(f"Channel(s) {sorted(overlap)} assigned to both roles; "
                      "put each channel in only one list.")
            return None, None
        if not ref and not seg:
            show_info("No channels selected.")
            return None, None
        return ref, seg

    # --- actions -----------------------------------------------------------
    def _scan_channels(self):
        input_path = self._input_path()
        if input_path is None:
            return
        chans = ChannelScanner(input_path).available_channels()
        if not chans:
            show_info("No channels with images found under single_channel/.")
        self.form.reference_channels.choices = chans
        self.form.segmentation_channels.choices = chans

    def _add_image_layer(self, input_path, chan, idx, opacity):
        try:
            arr = load_grayscale(resolve_image(input_path, chan, idx))
        except IndexError as e:
            show_info(str(e))
            return False
        self.viewer.add_image(
            arr, name=f"{chan} channel",
            colormap=DISPLAY_CMAP.get(chan, "gray"), opacity=opacity,
        )
        return True

    def _add_tracing_layers(self, seg_type, seg_channels, n_probes, point_size):
        if seg_type in PROBE_SEG_TYPES:
            for i in range(n_probes):
                self.viewer.add_points(name=f"{seg_type}_{i}", size=point_size)
        elif seg_type in SHAPE_SEG_TYPES:
            for chan in seg_channels:
                self.viewer.add_shapes(name=chan)
        else:  # POINT_SEG_TYPES
            for chan in seg_channels:
                self.viewer.add_points(name=chan, size=point_size)

    def _load_into_viewer(self):
        input_path = self._input_path()
        if input_path is None:
            return
        ref, seg = self._selected_channels()
        if ref is None:
            return
        idx = int(self.form.image_idx.value)
        point_size = int(self.form.point_size.value)
        for chan in ref:
            self._add_image_layer(input_path, chan, idx, REFERENCE_OPACITY)
        for chan in seg:
            self._add_image_layer(input_path, chan, idx, SEGMENTATION_OPACITY)
        self._add_tracing_layers(
            self.form.seg_type.value, seg, int(self.form.n_probes.value), point_size
        )

    def _save_segmentation(self):
        input_path = self._input_path()
        if input_path is None:
            return
        ref, seg = self._selected_channels()
        if ref is None:
            return
        seg_type = self.form.seg_type.value
        idx = int(self.form.image_idx.value)
        ref_for_stem = (seg + ref)
        if not ref_for_stem:
            show_info("No channels selected.")
            return
        stem = image_stem(input_path, ref_for_stem[0], idx)

        if seg_type in PROBE_SEG_TYPES:
            ids = [f"{seg_type}_{i}" for i in range(int(self.form.n_probes.value))]
            for pid in ids:
                if pid in self.viewer.layers:
                    save_points_layer(self.viewer.layers[pid].data,
                                      input_path, seg_type, pid, stem)
        elif seg_type in SHAPE_SEG_TYPES:
            for chan in seg:
                if chan in self.viewer.layers:
                    save_shapes_layer(list(self.viewer.layers[chan].data),
                                      input_path, seg_type, chan, stem)
        else:
            if not seg:
                show_info("No segmentation channels selected to save.")
                return
            for chan in seg:
                if chan in self.viewer.layers:
                    save_points_layer(self.viewer.layers[chan].data,
                                      input_path, seg_type, chan, stem)
        show_info("Segmentation saved.")
